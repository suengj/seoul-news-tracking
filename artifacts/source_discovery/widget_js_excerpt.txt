<script>
    LAYER_POPUP.RIGHT_NEWS_TEMPLATE = {
        news : null,
        distMsg : null,
        
        // 재난문자
        fn_getDistMsg : () => {
            
            // 재난문자
            let toDate = "2026-07-13"
            let oDistMsg ="";
            let sDistMsg = "";
            let sDistTitle="";
            $.ajax({
                type     : "post",
                url      : G_S_CONTEXT_PATH + "/disstr/selectDisstrSms.do",
                async    : true,
                dataType : "json",
                success  : function(res) {
                    
                    // 재난문자 출력 영역 초기화
                    $("#distMsg").html("");
                    
                    // 재난문자 존재 여부 파악
                    if (!res.sms.length) {
                        oDistMsg += `<tr>`;
                        oDistMsg += `    <td class="tal">`;
                        oDistMsg += `        <i>`;
                        oDistMsg += `            <img src="/css/assets/images/speaker.svg" alt="재난문자 아이콘">`;
                        oDistMsg += `        </i>`;
                        oDistMsg += `    </td>`;
                        oDistMsg += `    <td class="tal">`;
                        oDistMsg += `        <div>`;
                        oDistMsg += `            <a href="#seoul_common_gnb">오늘 재난문자가 없습니다.</a>`;
                        oDistMsg += `            <p>` + toDate + `</p>`;
                        oDistMsg += `        </div>`;
                        oDistMsg += `    </td>`;
                        oDistMsg += `</tr>`;
                        $("#distMsg").html(oDistMsg);
                    }
                    else {
                        
                        // 재난문자 출력 영역 설정
                        for (let i = 0; i < res.sms.length; i++) {
                            if (!!res.sms[i].smsMsg.split("] ")[1]) {
                                sDistMsg   = res.sms[i].smsMsg.split("] ")[1];
                                sDistTitle = res.sms[i].smsMsg.split("] ")[0].split("[")[1];
                            }
                            else if (!!res.sms[i].smsMsg.split("]")[1]) {
                                sDistMsg   = res.sms[i].smsMsg.split("]")[1];
                                sDistTitle = res.sms[i].smsMsg.split("]")[0].split("[")[1];
                            }
                            else {
                                sDistMsg   = res.sms[i].smsMsg.split("[")[0];
                                sDistTitle = res.sms[i].smsMsg.split("[")[1].split("]")[0];
                            }
                            oDistMsg += `<tr>`;
                            oDistMsg += `    <td class="tal">`;
                            oDistMsg += `        <div style="line-height: 30px; text-align: center; height: 30px; font-size: 14px; border:1px solid #ce225f; color: white; background-color: #ce225f; white-space: nowrap; overflow: hidden;">`;
                            oDistMsg += `            <p>` + sDistTitle + `</p>`;
                            oDistMsg += `        </div>`;
                            oDistMsg += `    </td>`;
                            oDistMsg += `    <td class="tal" style="padding: 16px 5px;" >`;
                            oDistMsg += `        <div class="txt msgDefault">`;
                            oDistMsg += `            <a onclick="COMMON.fn_setDistMsg(this)" style="font-size: 14px;" href="#seoul_common_gnb">`;
                            oDistMsg += `                <b style="font-weight: bold;">` + res.sms[i].disstrDate.replaceAll("/","-") + `</b> ` + sDistMsg;
                            oDistMsg += `            </a>`;
                            oDistMsg += `        </div>`;
                            oDistMsg += `        <div class="msgDetail">`;
                            oDistMsg += `            <a onclick="COMMON.fn_setDistMsg(this)" style="font-size: 14px;" href="#seoul_common_gnb">`;
                            oDistMsg += `                <b style="font-weight: bold;">` + res.sms[i].disstrDate.replaceAll("/","-") + `</b>`;
                            oDistMsg += `                <br> 송출지역 : ` + res.sms[i].lctnNm;
                            oDistMsg += `                <br>` + sDistMsg;
                            oDistMsg += `            </a>`;
                            oDistMsg += `        </div>`;
                            oDistMsg += `    </td>`;
                            oDistMsg += `</tr>`;
                        }
                        
                        // 재난문자 출력
                        setTimeout(function () {
                            $("#distMsg").append(oDistMsg);
                            $(".msgDetail").hide();
                        }, 0);
                    }
                },
                error(code) {
                    oDistMsg += `<tr>`;
                    oDistMsg += `    <td class="tal">`;
                    oDistMsg += `        <i>`;
                    oDistMsg += `            <img src="/css/assets/images/speaker.svg" alt="재난문자 아이콘">`;
                    oDistMsg += `        </i>`;
                    oDistMsg += `    </td>`;
                    oDistMsg += `    <td class="tal">`;
                    oDistMsg += `        <div>`;
                    oDistMsg += `            <a href="#seoul_common_gnb">재난문자 서비스 점검중입니다. CODE : [` + code + `]</a>`;
                    oDistMsg += `            <p>` + toDate + `</p>`;
                    oDistMsg += `        </div>`;
                    oDistMsg += `    </td>`;
                    oDistMsg += `</tr>`;
                    $("#distMsg").html(oDistMsg);
                }
            });
        },
